def sha(raw):
    return hashlib.sha256(raw).hexdigest()


@pytest.fixture
def worker(tmp_path, monkeypatch):
    case = make_case(tmp_path)
    prep.initial_context(state=case.state)
    graph, admission = admit(case)
    prepared_context = prep.context(state=case.state)
    clone = case.state / "worker-native-owner.duckdb"
    with IntentRepository(clone) as intent:
        local.materialize_local_benchmark_plan(admission=admission, intent=intent)
    task = graph.tasks[0]
    workspace = tmp_path / "allocated-source-worktree"
    git(case.root, "worktree", "add", "--detach", "-q", str(workspace))
    with open_existing_native_owner(database=clone, checkout=case.root,
            state_dir=case.state / "worker-native-owner",
            repository_id=case.prepared["manifest"]["payload"]["repository_cid"],
            execution_routes={task.task_key: GROK_CODEX_EXECUTION_MODE}) as owner:
        original_task = owner.source.get_task(task.task_cid)
        rebound = rebind_full_context(prepared_state=case.state, admission=admission,
            server=owner.server, output=case.root / ".runtime/worker-source-rebound")
        assert rebound["terminal_semantic_scope"] == prepared_context["terminal_semantic_scope"]
        portal = PortalTask(task_id=task.task_key, canonical_task_cid=task.task_cid,
            title=original_task.body["title"], status="ready", completion="manual", priority="P2",
            track="implementation", outputs=case.selected[:1], validation=[], acceptance="Public structure",
            metadata={"database task cid": task.task_cid})
        daemon = TodoImplementationDaemon(todo_path=case.root / ".runtime/worker-source.todo.md",
            state_path=case.state / "worker-prompt/tasks.json",
            strategy_path=case.state / "worker-prompt/strategy.json",
            events_path=case.state / "worker-prompt/events.jsonl", repo_root=case.root, task_header_prefix="## TB-")
        daemon._task_context_nomination_bundle = rebound["context_bundle"]
        try:
            prompt = daemon._build_implementation_prompt(portal, attempt=1)
            daemon._persist_implementation_context_receipt(portal, 1)
            capsule = next((case.state / "worker-prompt").rglob("*context-capsule.json"))
            public = instruction.prepare_public_instruction_context(repository=case.root,
                admission=admission, task_cid=task.task_cid, source_path=profiles.INSTRUCTION,
                expected_source_sha256=sha((case.root / profiles.INSTRUCTION).read_bytes()))
            request = dict(artifact=Path(public["artifact"]), expected_sha256=public["sha256"],
                task_cid=task.task_cid, prompt=prompt, workspace=workspace, repository=case.root)
            monkeypatch.chdir(workspace)
            monkeypatch.setenv("CODEX_HOME", str(tmp_path / "authored-empty-codex-home"))
            yield SimpleNamespace(case=case, task=task, rebound=rebound, prompt=prompt,
                capsule=capsule, public=public, request=request, workspace=workspace)
            assert owner.source.get_task(task.task_cid) == original_task
        finally:
            daemon.close_event_runtime()


def invoke(worker, **options):
    return runner.run(prompt=worker.prompt, provider="codex_cli", model="pinned", timeout=1,
        max_output_tokens=128, semantic_repository=worker.case.root,
        public_instruction_artifact=worker.request["artifact"],
        public_instruction_sha256=worker.request["expected_sha256"],
        public_instruction_task_cid=worker.task.task_cid, **options)


def source_endpoint(worker, monkeypatch):
    """Reuse real semantic pins; explicitly author only descriptor/UID routing."""
    from ipfs_accelerate_py.agent_supervisor.runtime import container_worker_boundary as boundary

    encoded = codec.encode_semantic_router_prompt(prompt=worker.prompt, repository=worker.case.root)
    binding = client.semantic_binding(encoded, "python-whitespace@1")
    descriptor_path = worker.case.state / "authored-worker-source-descriptor.json"
    descriptor = dict(schema=client.DESCRIPTOR_SCHEMA, session_id="1" * 32,
        workspace=str(worker.workspace), boundary_manifest_sha256="b" * 64, **binding)

    def load(path, digest, *, workspace):
        assert path == descriptor_path and digest == "d" * 64 and workspace == worker.workspace
        return deepcopy(descriptor)

    monkeypatch.setattr(client, "load_descriptor", load)
    monkeypatch.setattr(boundary, "verify_container_worker_boundary",
        lambda **kwargs: {"authored_uid_boundary": True})
    table = encoded.table.to_dict()
    views = WorkerSourceViews(repository=worker.case.root, workspace=worker.workspace,
        source_hashes={name: row["sha256"] for name, row in table["source_manifest"].items()}, **binding)
    options = dict(container_boundary=worker.case.state / "authored-boundary.json",
        container_boundary_sha256="b" * 64, source_view_profile="python-whitespace@1",
        source_view_descriptor=descriptor_path, source_view_descriptor_sha256="d" * 64)
    return views, options


@pytest.mark.parametrize("metadata_view", ["legacy", "common-bindings-instruction@1"])
def test_signed_large_population_reaches_router_selected_reads_and_historical_audit(
        worker, provider, monkeypatch, record_property, metadata_view):
    state, observed = provider
    payload = json.loads(worker.request["artifact"].read_bytes())
    assert payload["schema"] == "supervisor-public-instruction@5"
    assert payload["terminal_task_profile"] == worker.case.declared
    assert len(payload["manifest"]["payload"]["sources"]) == 73
    block, loaded = instruction.load_public_instruction(**worker.request)
    assert loaded["schema"] == "supervisor-public-instruction-inclusion@5"
    assert loaded["manifest_signature_verified"] is True
    selected = loaded["terminal_semantic_scope"]
    assert selected["schema"] == "terminal-signed-semantic-worker-inclusion@1"
    assert (selected["selected_count"], selected["full_count"], selected["frontier_count"]) == (5, 73, 68)
    assert selected["program_count"] == 2 and selected["required_count"] == 4
    assert (selected["max_files"], selected["max_source_bytes"]) == (64, 1_000_000)
    assert all(selected[key] is False for key in (
        "native_claim_verified", "native_allocation_verified", "process_identity_verified",
        "proof_authority", "execution_authority", "completion_authority", "omission_authority"))
    views, options = source_endpoint(worker, monkeypatch)
    read_events = []

    def read_selected_originals():
        page = views.discovery(max_bytes=16384)
        assert {row["path"] for row in page["rows"]} == set(worker.rebound["terminal_semantic_scope"]["selected_paths"])
        assert page["exhausted"] is True
        for name in (profiles.PROFILE, worker.case.selected[0]):
            source = views.page(path=name, view="original", max_bytes=16384)
            raw = source["content"].encode() if source["encoding"] == "utf-8" else base64.b64decode(source["content"], validate=True)
            assert raw == (worker.case.root / name).read_bytes()
            assert sha(raw) == source["slice_sha256"]
            read_events.append(name)
        with pytest.raises(ValueError):
            views.page(path="module_069.py", view="original", max_bytes=16384)

    state["after_generate"] = read_selected_originals
    text, receipt = invoke(worker, semantic_metadata_view=metadata_view, **options)
    assert text == "literal summary" and len(observed) == 1
    assert read_events == [profiles.PROFILE, worker.case.selected[0]]
    assert receipt["terminal_semantic_scope_dispatch_rechecked"] is True
    assert receipt["public_instruction"]["terminal_semantic_scope"] == selected
    assert receipt["completion_authority"] is False
    assert receipt["source_view_retrieval"]["requests_observed"] is False
    if metadata_view != "legacy":
        assert receipt["semantic_metadata_view"]["requested_mode"] == metadata_view
    actual = observed[0][0]
    wire, _ = json.JSONDecoder().raw_decode(actual[receipt["workspace_advisory_bytes"]:])
    assert wire["translated_semantic"]["raw_sources"][profiles.PROFILE] == profiles.task_profile_bytes(worker.case.declared).decode()
    assert set(wire["translated_semantic"]["manifest"]) == set(worker.rebound["terminal_semantic_scope"]["selected_paths"])
    assert actual.count(block) == 1
    assert receipt["model_prompt_sha256"] == sha(actual.encode())
    historical, rendered, _ = audit._capsule(worker.capsule.read_bytes(), expected=worker.rebound,
        prepared=worker.case.prepared)
    assert historical["all_context_checks_passed"] is True and rendered == worker.prompt
    parsed = audit._receipt({**receipt, "phase": "coding"}, worker.workspace.parent)
    unchecked = deepcopy(receipt)
    unchecked["terminal_semantic_scope_dispatch_rechecked"] = False
    with pytest.raises(ValueError):
        audit._receipt({**unchecked, "phase": "coding"}, worker.workspace.parent)
    # Stored input replay remains historical after current originals drift and
    # the linked worker checkout is removed. It grants no fresh dispatch.
    (worker.case.root / "module_069.py").write_text("def later_state(): return 7\n")
    monkeypatch.chdir(worker.case.root.parent)
    git(worker.case.root, "worktree", "remove", "--force", str(worker.workspace))
    replayed, _, checks = audit._model_projection(rendered=worker.prompt, receipt=parsed,
        repository=worker.case.root)
    assert replayed == actual and all(checks.values()), checks
    for key, value in (("profile_sha256", "0" * 64), ("frontier_cid", selected["declaration_cid"])):
        changed = deepcopy(parsed)
        changed["public_instruction"]["terminal_semantic_scope"][key] = value
        _, _, failed_checks = audit._model_projection(rendered=worker.prompt, receipt=changed,
            repository=worker.case.root)
        assert not all(failed_checks.values())
    record_property("authored_provider_fixture_calls", 1)
    record_property("external_provider_calls", 0)
    record_property("native_coding_child_launched", False)
    record_property("official_verifier_executed", False)
    record_property("provider_token_savings_measured", False)
    record_property("semantic_metadata_view_requested", metadata_view)


@pytest.mark.parametrize("location,source", [
    ("canonical", "selected"), ("allocated", "selected"),
    ("canonical", "omitted"), ("allocated", "omitted"),
    ("canonical", "support"), ("allocated", "support"),
    ("canonical", "profile"), ("allocated", "profile"),
])
def test_provider_initialization_drift_refuses_before_generate_text(worker, provider, monkeypatch, capsys, location, source):
    from ipfs_accelerate_py import llm_router

    _, observed = provider
    calls = []
    name = {"selected": worker.case.selected[0], "omitted": "module_069.py",
        "support": profiles.SMOKE, "profile": profiles.PROFILE}[source]
    root = worker.case.root if location == "canonical" else worker.workspace

    def initialize(*args, **kwargs):
        calls.append(True)
        path = root / name
        path.write_bytes(path.read_bytes() + b"\n")
        return object()

    monkeypatch.setattr(llm_router, "get_llm_provider", initialize)
    capsys.readouterr()
    with pytest.raises(ValueError):
        invoke(worker)
    receipt = json.loads(capsys.readouterr().out)
    assert receipt["schema"] == "router-implementation-invocation@1"
    assert receipt["status"] == "failed"
    assert receipt["failure_phase"] == "terminal_semantic_scope_currentness"
    assert receipt["terminal_semantic_scope_dispatch_rechecked"] is False
    assert calls == [True]
    assert observed == []
    assert len(worker.case.prepared["manifest"]["payload"]["sources"]) == 73
    assert not (worker.case.state / "planner-invoked.json").exists()
